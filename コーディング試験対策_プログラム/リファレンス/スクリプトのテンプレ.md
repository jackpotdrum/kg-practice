# スクリプトのテンプレ

どの問題でも使い回しやすい、入力処理とバリデーションのテンプレです。

```python
import argparse


def validate_input(length):
    # バリデーション
    errors = []
    if not (1 <= length <= 2 * 10 ** 5):
        errors.append("長さは1以上2×10^5以下の整数でなければなりません")
    return errors


def parse_args():
    # 入力引数の定義
    parser = argparse.ArgumentParser(description="スクリプト説明文")
    parser.add_argument("--length", type=int)
    return parser.parse_args()


def main():
    args = parse_args()
    length = args.length

    errors = validate_input(length)
    if errors:
        print("入力に誤りがあります")
        print("======================================")
        for error in errors:
            print(error)
        return

    # 本処理


if __name__ == "__main__":
    main()
```
